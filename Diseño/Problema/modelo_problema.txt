package DAO {

class UsuarioDAO {
+ registrar(usuario)
+ login(email,password)
+ obtenerPorId(id)
+ actualizar(usuario)
+ cambiarPassword(id,pass)
}

class RutinaDAO {
+ obtenerTodas()
+ obtenerPorNivel(nivel)
+ obtenerPorId(id)
+ obtenerEntrenamientos(id)
+ asignarAUsuario(idUser,idRut)
}

class EjercicioDAO {
+ obtenerTodos()
+ obtenerPorMusculo(musculo)
+ obtenerPorId(id)
}

class RecetaDAO {
+ obtenerTodas()
+ obtenerPorCategoria(cat)
+ obtenerPorId(id)
+ agregarFavorita(idUser,idRec)
+ obtenerFavoritas(idUser)
}

class RegistroDAO {
+ registrarEntrenamiento(reg)
+ obtenerHistorial(idUser)
+ registrarAgua(reg)
+ obtenerConsumoAgua(idUser,fecha)
}

class ProgresoDAO {
+ obtenerProgresoSemanal(id,sem)
+ crearObjetivo(objetivo)
+ obtenerObjetivos(idUser)
}

}
